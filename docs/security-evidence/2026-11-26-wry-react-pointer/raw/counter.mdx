import {useEffect, useState} from 'react';

export function Counter() {
  const [count, setCount] = useState(0);
  useEffect(() => {
    document.title = JSON.stringify({ready:true, count, field:'', focus:document.activeElement.id, error:globalThis.__pnFixtureError || ''});
  }, [count]);
  return <><button id="counter" onClick={() => setCount(value => value + 1)}>Count: {count}</button><p>PINNED MDX REACT HOOKS</p></>;
}

<Counter />
