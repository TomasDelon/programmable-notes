function keys(pid, sequence) {
  const ids = execFileSync(
    'xdotool',
    ['search', '--onlyvisible', '--pid', String(pid)],
    { encoding: 'utf8', timeout: 10000 },
  )
    .trim()
    .split(/\s+/);
  assert.equal(ids.length, 1);
  execFileSync(
    'xdotool',
    ['windowactivate', '--sync', ids[0], 'key', '--clearmodifiers', sequence],
    { timeout: 10000 },
  );
}
async function disk(value) {
  assert.equal(await readFile(path, 'utf8'), value);
  assert.equal((await stat(path)).mode & 0o777, 0o640);
}
const startTick = async (pid) => {
  const record = await readFile(`/proc/${pid}/stat`, 'utf8');
  return record.slice(record.lastIndexOf(') ') + 2).split(' ')[19];
};
let ownedRoot;
// WebDriver deletion can leave its Rust root alive after a renderer crash.
// Touch only this session's captured process, never an executable-name search.
async function cleanupOwnedRoot() {
  if (!ownedRoot) return;
  const { pid, start } = ownedRoot;
  const gone = (error) => error.code === 'ENOENT' || error.code === 'ESRCH';
  try {
    if ((await startTick(pid)) !== start) return;
  } catch (error) {
    if (gone(error)) return;
    throw error;
  }
  assert.equal(await rootPid(), pid);
  assert.equal(
    digest(await readFile(`/proc/${pid}/exe`)),
    report.executableSha256,
  );
  try {
    if ((await startTick(pid)) !== start) return;
    process.kill(pid, 'SIGTERM');
  } catch (error) {
    if (gone(error)) return;
    throw error;
  }
  for (let i = 0; i < 100; i++) {
    try {
      if ((await startTick(pid)) !== start) {
        report.ownedRootTerminatedAfterSession = true;
        return;
      }
    } catch (error) {
      if (gone(error)) {
        report.ownedRootTerminatedAfterSession = true;
        return;
      }
      throw error;
    }
    await pause(50);
  }
  throw Error('Owned root did not exit');
}
