/** Baseline observation, not new native lifecycle acceptance. */
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {readFile,writeFile} from 'node:fs/promises';
import {execFileSync} from 'node:child_process';
import {join} from 'node:path';
import process from 'node:process';
import {preflight,launch,close,script,invoke,systemctl} from './native-webdriver-lib.mjs';
import {rootPid,waitUi,button,dialog,fileIdle,pause} from './native-ui-lib.mjs';
await preflight();assert.equal(process.env.NATIVE_DISPOSABLE_PROFILE,'1');
const source="import React from 'react'\n\nexport function SessionProbe(){const[n,setN]=React.useState(0);return <button onClick={()=>setN(n+1)}>Session count {n}</button>}\n\n<SessionProbe />\n";
const file=join(process.env.NATIVE_FIXTURE_DIR,'session-status.mdx');await writeFile(file,source,{flag:'wx',mode:0o640});
const report={scope:'baseline actual UI across natural expiry; no new API',observerSha256:createHash('sha256').update(await readFile(new URL(import.meta.url))).digest('hex')};
const output=join(process.env.NATIVE_EVIDENCE_DIR,'session-baseline.json');let failure;
const ui=()=>script(`const done=arguments[arguments.length-1];done({build:document.querySelector('.build-status')?.textContent,session:document.querySelector('[aria-label="Native preview status"]')?.textContent??null,nativeNote:Array.from(document.querySelectorAll('p[role="note"]')).map(e=>e.textContent).filter(v=>v.includes('separate resource-limited')),alerts:Array.from(document.querySelectorAll('[role="alert"]')).map(e=>e.textContent)});`);
const list=(pid)=>systemctl('list-units','--all','--plain','--no-legend',`programmable-notes-preview-${pid}-*.service`).split('\n').filter(Boolean).map(row=>row.trim().split(/\s+/)[0]);
try{
 await launch();const pid=await rootPid();report.editorPid=pid;report.executableSha256=createHash('sha256').update(await readFile(`/proc/${pid}/exe`)).digest('hex');
 await waitUi("!!document.querySelector('.cm-content')");await button('Stop preview');assert.equal((await invoke('stop_preview')).ok,true);
 await button('Pause automatic recovery');await button('Open files');dialog(pid,'choose','Open documents and modules',file);await fileIdle();
 const old=new Set(list(pid));await button('Rebuild preview');
 await waitUi("document.querySelector('.build-status')?.textContent.startsWith('Compiled')");
 for(let i=0;i<200;i++){
  const candidate=list(pid).find(unit=>!old.has(unit));
  if(candidate){const p=Number(systemctl('show','--property=MainPID','--value',candidate));if(p>0){report.unit=candidate;report.previewPid=p;break;}}
  await pause(100);
 }
 assert.ok(report.unit,'No new preview unit');
 report.policy=systemctl('show',report.unit,'--property=MemoryMax,MemorySwapMax,CPUQuotaPerSecUSec,TasksMax,RuntimeMaxUSec,PrivateNetwork,ControlGroup');
 execFileSync('python3',['scripts/native-atspi.py',String(report.previewPid),'Session count 0','click'],{timeout:65000});
 execFileSync('python3',['scripts/native-atspi.py',String(report.previewPid),'Session count 1','find'],{timeout:65000});
 report.before=await ui();assert.equal(report.before.session,null);
 const deadline=Date.now()+80000;
 while(Date.now()<deadline){try{execFileSync('systemctl',['--user','is-active','--quiet',report.unit],{stdio:'pipe',timeout:10000});}catch(error){if(error.status===3||error.status===4){report.observedInactive=true;break;}throw error;}await pause(250);}
 assert.equal(report.observedInactive,true);
 report.journal=execFileSync('journalctl',['--user','-u',report.unit,'--no-pager','-o','short-iso'],{encoding:'utf8',timeout:10000});
 assert.match(report.journal,/Service reached runtime time limit/);
 await pause(1500);report.after=await ui();assert.equal(report.after.session,null);
 assert.equal(await readFile(file,'utf8'),source);report.sourceUnchanged=true;report.outcome='observed';
}catch(error){failure=error;report.error=String(error);report.outcome='fail';}
finally{
 await invoke('stop_preview').catch(error=>{report.stopError=String(error)});await close().catch(error=>{report.closeError=String(error)});
 await writeFile(output,JSON.stringify(report,null,2)+'\n');console.log('NATIVE_SESSION_BASELINE',output);
}
if(failure)throw failure;
