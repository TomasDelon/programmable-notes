# External disposable-VM administrator diagnostic; never loaded by the app.
import hashlib,json,os,re,signal,subprocess,sys,tempfile,time
from pathlib import Path
fixture=Path(sys.argv[1]);assert fixture.parent==Path('/tmp') and re.fullmatch(r'pn-native-[A-Za-z0-9]+',fixture.name)
evidence=fixture/'evidence';barrier=evidence/'kill-editor-renderer.json'
expected=sys.argv[2];assert re.fullmatch(r'[a-f0-9]{64}',expected)
def state(pid):
    p=Path('/proc')/str(pid);s=(p/'stat').read_text().rsplit(') ',1)[1].split()
    fields={line.split(':',1)[0]:line.split(':',1)[1].strip() for line in (p/'status').read_text().splitlines() if ':' in line}
    return {'pid':pid,'startTick':s[19],'parent':int(s[1]),'exe':os.readlink(p/'exe'),'uid':int(fields['Uid'].split()[0]),'namespacePids':[int(v) for v in fields['NSpid'].split()],'tracer':int(fields['TracerPid'])}
def descends(pid,parent):
    seen=set()
    while pid>1 and pid not in seen:
        if pid==parent:return True
        seen.add(pid)
        try:pid=state(pid)['parent']
        except (FileNotFoundError,ProcessLookupError):return False
    return False
end=time.monotonic()+300
while not barrier.exists():
    assert time.monotonic()<end,'Observer did not request owned renderer termination'
    time.sleep(.1)
b=json.loads(barrier.read_text());root=state(int(b['pid']))
assert root['uid']==1000 and root['exe']=='/usr/bin/programmable-notes'
assert root['startTick']==b['startTick']
assert hashlib.sha256(Path('/proc',str(root['pid']),'exe').read_bytes()).hexdigest()==b['executableSha256']==expected
assert int((fixture/'editor.pid').read_text())==root['pid']
candidates=[]
for path in Path('/proc').iterdir():
    if not path.name.isdigit():continue
    try:
        p=state(int(path.name))
        if p['exe']=='/usr/lib/x86_64-linux-gnu/webkit2gtk-4.1/WebKitWebProcess' and p['uid']==1000 and descends(p['pid'],root['pid']):candidates.append(p)
    except (FileNotFoundError,ProcessLookupError,PermissionError):pass
print('EDITOR_KILL_OWNED_CANDIDATES',json.dumps({'root':root,'candidates':candidates}),flush=True)
assert len(candidates)==1,'Require one owned Root renderer, never choose by name alone'
target=candidates[0];assert target['tracer']==0 and target['pid']>1
targetMount=os.readlink(f'/proc/{target["pid"]}/ns/mnt')
targetPidNamespace=os.readlink(f'/proc/{target["pid"]}/ns/pid')
assert state(target['pid'])==target

assert b'--preview-child' not in Path('/proc',str(root['pid']),'cmdline').read_bytes()
assert state(target['pid'])==target
rendererHash=hashlib.sha256(Path('/proc',str(target['pid']),'exe').read_bytes()).hexdigest()
os.kill(target['pid'],signal.SIGKILL)
end=time.monotonic()+10
while Path('/proc',str(target['pid'])).exists():
    try:
        status=Path('/proc',str(target['pid']),'stat').read_text().rsplit(') ',1)[1].split()[0]
        if status=='Z':break
    except FileNotFoundError:break
    assert time.monotonic()<end,'Owned renderer did not terminate'
    time.sleep(.1)
assert state(root['pid'])['startTick']==root['startTick'],'Root must survive renderer kill'
receipt=json.dumps({'root':{**root,'executableSha256':expected},'renderer':{**target,'executableSha256':rendererHash},'signal':'SIGKILL','rootSurvived':True})
writer='import os,sys;f=os.open(sys.argv[1],os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600);os.write(f,sys.argv[2].encode());os.close(f)'
subprocess.run(['runuser','-u','ubuntu','--','python3','-c',writer,str(evidence/'editor-renderer-killed.json'),receipt],check=True)
print('EDITOR_RENDERER_IDENTITY_CHECKED_KILLED',receipt,flush=True)
